# Exemple: health check simple
import subprocess, time
HOSTS = ["192.168.10.1", "192.168.10.10"]
while True:
    for h in HOSTS:
        ok = subprocess.call(["ping", "-c", "1", h], stdout=subprocess.DEVNULL)==0
        print(time.strftime('%H:%M:%S'), h, "UP" if ok else "DOWN")
    time.sleep(30)
